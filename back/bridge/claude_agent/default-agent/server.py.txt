"""OpenAI-compatible HTTP adapter around the Claude Agent SDK.

This file is shipped into the isolated runtime image. Galaris consumes standard
Chat Completions chunks and two optional named SSE events carrying normalized tool
messages and terminal telemetry.
"""

from __future__ import annotations

import asyncio
import hmac
import json
import os
from pathlib import Path
import time
from typing import Any, AsyncIterator
from uuid import uuid4

from claude_agent_sdk import (
    ClaudeAgentOptions,
    HookMatcher,
    PermissionResultAllow,
    PermissionResultDeny,
    query,
)
from claude_agent_sdk.types import ResultMessage, StreamEvent
from fastapi import FastAPI, HTTPException, Request
from fastapi.responses import JSONResponse, StreamingResponse
from pydantic import BaseModel, ConfigDict, Field
from stream_trace import ClaudeStreamTrace


class ChatMessage(BaseModel):
    model_config = ConfigDict(extra="ignore")

    role: str
    content: Any = None


class ChatRequest(BaseModel):
    model_config = ConfigDict(extra="ignore")

    model: str
    messages: list[ChatMessage] = Field(default_factory=list)
    stream: bool = False


app = FastAPI(title="Claude Agent Harness", docs_url=None, redoc_url=None)


def _authenticate(request: Request) -> None:
    expected = os.environ.get("CLAUDE_AGENT_API_TOKEN", "")
    supplied = request.headers.get("Authorization", "").removeprefix("Bearer ").strip()
    if not expected or not hmac.compare_digest(expected, supplied):
        raise HTTPException(status_code=401, detail="Invalid Harness token")


def _header_int(request: Request, name: str, default: int, maximum: int) -> int:
    try:
        value = int(request.headers.get(name, "") or default)
    except ValueError:
        value = default
    return max(1, min(value, maximum))


def _content_text(value: Any) -> str:
    if isinstance(value, str):
        return value
    if not isinstance(value, list):
        return "" if value is None else str(value)
    parts: list[str] = []
    for item in value:
        if isinstance(item, dict) and item.get("type") == "text":
            parts.append(str(item.get("text") or ""))
        elif isinstance(item, str):
            parts.append(item)
    return "".join(parts)


def _prompt(messages: list[ChatMessage]) -> tuple[str, str]:
    system_parts: list[str] = []
    turns: list[str] = []
    for message in messages:
        content = _content_text(message.content).strip()
        if not content:
            continue
        role = message.role.strip().lower()
        if role == "system":
            system_parts.append(content)
        elif role in {"user", "assistant"}:
            turns.append(f"<{role}>\n{content}\n</{role}>")
    return "\n\n".join(system_parts), "\n\n".join(turns)


def _bounded_json(value: Any, limit: int = 8_000) -> dict[str, Any]:
    try:
        encoded = json.dumps(value, ensure_ascii=False, default=str)
    except Exception:
        encoded = str(value)
    if len(encoded) > limit:
        encoded = encoded[: limit - 20] + "…[truncated]"
    try:
        decoded = json.loads(encoded)
    except json.JSONDecodeError:
        return {"output": encoded}
    return decoded if isinstance(decoded, dict) else {"output": decoded}


def _usage(result: ResultMessage, tool_calls: int) -> dict[str, Any]:
    input_tokens = 0
    output_tokens = 0
    cache_read_tokens = 0
    cache_write_tokens = 0
    cost = 0.0
    model_usage = result.model_usage or {}
    for value in model_usage.values():
        if not isinstance(value, dict):
            continue
        input_tokens += int(value.get("inputTokens") or 0)
        output_tokens += int(value.get("outputTokens") or 0)
        cache_read_tokens += int(value.get("cacheReadInputTokens") or 0)
        cache_write_tokens += int(value.get("cacheCreationInputTokens") or 0)
        cost += float(value.get("costUSD") or 0.0)
    if not model_usage and isinstance(result.usage, dict):
        input_tokens = int(result.usage.get("input_tokens") or 0)
        output_tokens = int(result.usage.get("output_tokens") or 0)
        cache_read_tokens = int(result.usage.get("cache_read_input_tokens") or 0)
        cache_write_tokens = int(result.usage.get("cache_creation_input_tokens") or 0)
    if cost <= 0 and result.total_cost_usd:
        cost = float(result.total_cost_usd)
    return {
        "input_tokens": max(0, input_tokens),
        "output_tokens": max(0, output_tokens),
        "cache_read_tokens": max(0, cache_read_tokens),
        "cache_write_tokens": max(0, cache_write_tokens),
        "requests": max(0, int(result.num_turns)),
        "tool_calls": max(0, tool_calls),
        "cost": max(0.0, cost),
        "token_quality": "partial",
        "cost_quality": "estimated" if cost > 0 else "unknown",
    }


def _sse(event: str | None, payload: Any) -> str:
    prefix = f"event: {event}\n" if event else ""
    return prefix + "data: " + json.dumps(payload, ensure_ascii=False, default=str) + "\n\n"


async def _prompt_messages(prompt: str) -> AsyncIterator[dict[str, Any]]:
    """Use SDK streaming input so the tool permission callback remains available."""

    yield {
        "type": "user",
        "message": {"role": "user", "content": prompt},
    }


async def _run_events(
    body: ChatRequest,
    request: Request,
) -> AsyncIterator[tuple[str, dict[str, Any]]]:
    system_prompt, prompt = _prompt(body.messages)
    if not prompt:
        raise HTTPException(status_code=422, detail="No user message was provided")

    workspace = Path(os.environ.get("CLAUDE_AGENT_WORKSPACE", "/data/workspace"))
    workspace.mkdir(parents=True, exist_ok=True)
    task_id = request.headers.get("X-Galaris-Task-Id", "").strip()
    run_id = request.headers.get("X-Galaris-Agent-Run-Id", "").strip()
    model = request.headers.get("X-Galaris-Model-Code", "").strip() or body.model
    reasoning_effort = request.headers.get(
        "X-Galaris-Reasoning-Effort", ""
    ).strip().lower()
    approval = request.headers.get("X-Galaris-Approval-Action", "ask").strip()
    max_turns = _header_int(request, "X-Galaris-Request-Limit", 40, 500)
    tool_limit = _header_int(request, "X-Galaris-Tool-Call-Limit", 80, 2_000)

    semantic_events: asyncio.Queue[dict[str, Any]] = asyncio.Queue()
    tool_started: dict[str, float] = {}
    tool_count = 0

    async def pre_tool(input_data: dict[str, Any], tool_use_id: str | None, _context: Any):
        nonlocal tool_count
        call_id = str(input_data.get("tool_use_id") or tool_use_id or uuid4())
        tool_started[call_id] = time.perf_counter()
        tool_count += 1
        if tool_count > tool_limit:
            return {
                "hookSpecificOutput": {
                    "hookEventName": "PreToolUse",
                    "permissionDecision": "deny",
                    "permissionDecisionReason": "Galaris tool-call limit reached",
                }
            }
        return {}

    async def post_tool(input_data: dict[str, Any], tool_use_id: str | None, _context: Any):
        call_id = str(input_data.get("tool_use_id") or tool_use_id or "")
        elapsed = max(0.0, time.perf_counter() - tool_started.pop(call_id, time.perf_counter()))
        response = input_data.get("tool_response")
        message = {
            "type": "tool",
            "content": str(response)[:4_000] if response is not None else "Tool completed",
            "tool_name": str(input_data.get("tool_name") or "unknown"),
            "tool_arguments": _bounded_json(input_data.get("tool_input") or {}),
            "tool_result": _bounded_json(response),
            "execution_time": elapsed,
            "cost": 0.0,
            "success": True,
        }
        await semantic_events.put(message)
        return {}

    async def failed_tool(input_data: dict[str, Any], tool_use_id: str | None, _context: Any):
        call_id = str(input_data.get("tool_use_id") or tool_use_id or "")
        elapsed = max(0.0, time.perf_counter() - tool_started.pop(call_id, time.perf_counter()))
        error = str(input_data.get("error") or "Tool failed")
        await semantic_events.put(
            {
                "type": "tool",
                "content": error[:4_000],
                "tool_name": str(input_data.get("tool_name") or "unknown"),
                "tool_arguments": _bounded_json(input_data.get("tool_input") or {}),
                "tool_result": {"error": error[:8_000]},
                "execution_time": elapsed,
                "cost": 0.0,
                "success": False,
            }
        )
        return {}

    async def permission(tool_name: str, input_data: dict[str, Any], _context: Any):
        if approval == "auto":
            return PermissionResultAllow(updated_input=input_data)
        reason = (
            "An AI requester cannot approve this operation."
            if approval == "deny_agent"
            else "This runtime cannot pause a Chat Completions request for human approval; enable Task auto-approval to permit it."
        )
        await semantic_events.put(
            {
                "type": "tool",
                "content": reason,
                "tool_name": "approval",
                "tool_arguments": {"requested_tool": tool_name},
                "tool_result": {"decision": "deny"},
                "execution_time": 0.0,
                "cost": 0.0,
                "success": False,
            }
        )
        return PermissionResultDeny(message=reason, interrupt=False)

    custom_headers = []
    if task_id:
        custom_headers.append(f"X-Galaris-Task-Id: {task_id}")
    if run_id:
        custom_headers.append(f"X-Galaris-Agent-Run-Id: {run_id}")
    if reasoning_effort:
        custom_headers.append(
            f"X-Galaris-Reasoning-Effort: {reasoning_effort}"
        )
    sdk_env = {
        "ANTHROPIC_CUSTOM_HEADERS": "\n".join(custom_headers),
        "ANTHROPIC_CUSTOM_MODEL_OPTION": model,
        "ANTHROPIC_DEFAULT_OPUS_MODEL": "standard",
        "ANTHROPIC_DEFAULT_FABLE_MODEL": "high",
        "ANTHROPIC_DEFAULT_SONNET_MODEL": "low",
        "ANTHROPIC_DEFAULT_HAIKU_MODEL": "ultra-low",
        "API_TIMEOUT_MS": "600000",
        "CLAUDE_CODE_MAX_RETRIES": "2",
    }
    mcp_url = os.environ.get("GALARIS_MCP_URL", "").strip()
    mcp_token = os.environ.get("GALARIS_MCP_TOKEN", "").strip()
    mcp_servers = {}
    if mcp_url and mcp_token:
        mcp_servers["galaris"] = {
            "type": "http",
            "url": mcp_url,
            "headers": {"Authorization": f"Bearer {mcp_token}"},
        }
    options = ClaudeAgentOptions(
        tools={"type": "preset", "preset": "claude_code"},
        system_prompt={
            "type": "preset",
            "preset": "claude_code",
            "append": system_prompt,
        },
        mcp_servers=mcp_servers,
        strict_mcp_config=bool(mcp_servers),
        permission_mode="bypassPermissions" if approval == "auto" else "default",
        can_use_tool=permission,
        hooks={
            "PreToolUse": [HookMatcher(hooks=[pre_tool])],
            "PostToolUse": [HookMatcher(hooks=[post_tool])],
            "PostToolUseFailure": [HookMatcher(hooks=[failed_tool])],
        },
        include_partial_messages=True,
        model=model,
        cwd=workspace,
        add_dirs=[Path("/data/skills")],
        setting_sources=["project"],
        max_turns=max_turns,
        effort={
            "none": "low",
            "minimal": "low",
            "low": "low",
            "medium": "medium",
            "high": "high",
            "xhigh": "max",
            "max": "max",
        }.get(reasoning_effort),
        env=sdk_env,
    )

    started_at = time.perf_counter()
    emitted_text = ""
    terminal: ResultMessage | None = None
    trace = ClaudeStreamTrace()
    try:
        async for message in query(prompt=_prompt_messages(prompt), options=options):
            while not semantic_events.empty():
                yield "tool", semantic_events.get_nowait()
            if isinstance(message, StreamEvent):
                event = message.event
                for thinking in trace.consume(event):
                    yield "tool", thinking
                if event.get("type") != "content_block_delta":
                    continue
                delta = event.get("delta") or {}
                if delta.get("type") == "text_delta":
                    content = str(delta.get("text") or "")
                    if content:
                        emitted_text += content
                        yield "text", {"content": content}
            elif isinstance(message, ResultMessage):
                terminal = message
        while not semantic_events.empty():
            yield "tool", semantic_events.get_nowait()
        for thinking in trace.finish():
            yield "tool", thinking
    except Exception as exc:
        yield "result", {
            "success": False,
            "execution_time": max(0.0, time.perf_counter() - started_at),
            "usage": {
                "requests": 0,
                "tool_calls": tool_count,
                "token_quality": "unknown",
                "cost_quality": "unknown",
            },
            "metadata": {
                "runtime": "claude-agent-sdk",
                "runtime_status": "failed",
                "error": f"{type(exc).__name__}: {exc}"[:4_000],
            },
        }
        return

    if terminal is None:
        yield "result", {
            "success": False,
            "execution_time": max(0.0, time.perf_counter() - started_at),
            "usage": {
                "requests": 0,
                "tool_calls": tool_count,
                "token_quality": "unknown",
                "cost_quality": "unknown",
            },
            "metadata": {
                "runtime": "claude-agent-sdk",
                "runtime_status": "failed",
                "error": "Claude Agent emitted no terminal result",
            },
        }
        return

    final_text = terminal.result or ""
    if final_text and final_text != emitted_text:
        missing = final_text[len(emitted_text):] if final_text.startswith(emitted_text) else final_text
        if missing:
            yield "text", {"content": missing}
    yield "result", {
        "success": not terminal.is_error,
        "result": final_text,
        "execution_time": max(0.0, time.perf_counter() - started_at),
        "usage": _usage(terminal, tool_count),
        "metadata": {
            "runtime": "claude-agent-sdk",
            "runtime_run_id": terminal.session_id,
            "session_id": terminal.session_id,
            "runtime_status": terminal.subtype,
            "stop_reason": terminal.stop_reason,
            "terminal_reason": terminal.terminal_reason,
            "num_turns": terminal.num_turns,
            "errors": list(terminal.errors or [])[:20],
            "permission_denials": list(terminal.permission_denials or [])[:50],
            "model_usage": terminal.model_usage or {},
        },
    }


@app.get("/health")
async def health() -> dict[str, str]:
    return {"status": "ok", "runtime": "claude-agent-sdk"}


@app.get("/v1/models")
async def models(request: Request) -> dict[str, Any]:
    _authenticate(request)
    return {
        "object": "list",
        "data": [
            {
                "id": "claude-agent",
                "object": "model",
                "owned_by": "galaris",
            }
        ],
    }


@app.post("/v1/chat/completions")
async def completions(body: ChatRequest, request: Request):
    _authenticate(request)
    completion_id = f"chatcmpl_{uuid4().hex}"
    if body.stream:
        async def stream() -> AsyncIterator[str]:
            async for kind, payload in _run_events(body, request):
                if kind == "text":
                    yield _sse(
                        None,
                        {
                            "id": completion_id,
                            "object": "chat.completion.chunk",
                            "model": body.model,
                            "choices": [
                                {
                                    "index": 0,
                                    "delta": {"content": payload["content"]},
                                    "finish_reason": None,
                                }
                            ],
                        },
                    )
                elif kind == "tool":
                    yield _sse("galaris.agent-message/v1", payload)
                else:
                    yield _sse("galaris.agent-result/v1", payload)
                    usage = payload.get("usage") or {}
                    yield _sse(
                        None,
                        {
                            "id": completion_id,
                            "object": "chat.completion.chunk",
                            "model": body.model,
                            "choices": [],
                            "usage": {
                                "prompt_tokens": usage.get("input_tokens", 0),
                                "completion_tokens": usage.get("output_tokens", 0),
                                "total_tokens": usage.get("input_tokens", 0)
                                + usage.get("output_tokens", 0),
                            },
                        },
                    )
            yield "data: [DONE]\n\n"

        return StreamingResponse(
            stream(),
            media_type="text/event-stream",
            headers={
                "Cache-Control": "no-cache",
                "X-Accel-Buffering": "no",
            },
        )

    text_parts: list[str] = []
    tool_messages: list[dict[str, Any]] = []
    terminal: dict[str, Any] = {
        "success": False,
        "usage": {"token_quality": "unknown", "cost_quality": "unknown"},
        "metadata": {"runtime_status": "failed"},
    }
    async for kind, payload in _run_events(body, request):
        if kind == "text":
            text_parts.append(str(payload.get("content") or ""))
        elif kind == "tool":
            tool_messages.append(payload)
        else:
            terminal = payload
    content = "".join(text_parts)
    usage = terminal.get("usage") or {}
    return JSONResponse(
        {
            "id": completion_id,
            "object": "chat.completion",
            "model": body.model,
            "choices": [
                {
                    "index": 0,
                    "message": {"role": "assistant", "content": content},
                    "finish_reason": "stop" if terminal.get("success") else "error",
                }
            ],
            "usage": {
                "prompt_tokens": usage.get("input_tokens", 0),
                "completion_tokens": usage.get("output_tokens", 0),
                "total_tokens": usage.get("input_tokens", 0)
                + usage.get("output_tokens", 0),
            },
            "galaris": {
                **terminal,
                "messages": tool_messages,
            },
        }
    )
