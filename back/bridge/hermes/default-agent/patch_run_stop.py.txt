"""Add worker-exit evidence to the pinned Hermes run protocol at image build time.

Upstream cancellation can finish the asyncio task before its executor thread exits.
Only that thread's return may prove that the run can no longer execute agent work.
No cancellation, scheduling or result behavior is changed here.
"""

from pathlib import Path

path = Path("/opt/hermes/gateway/platforms/api_server_runs.py")
source = path.read_text()
original = """        result, usage = await loop.run_in_executor(
            None, lambda: _run_agent_sync(self, run, agent, approval_notify, _api_server=_api_server))"""
replacement = """        def mark_execution_stopped():
            self._galaris_authorization_loops.pop(run_id, None)
            current = self._run_statuses.get(run_id)
            if current is not None:
                self._set_run_status(run_id, current['status'], execution_stopped=True)

        def tracked_execution():
            try:
                return _run_agent_sync(self, run, agent, approval_notify, _api_server=_api_server)
            finally:
                loop.call_soon_threadsafe(mark_execution_stopped)

        self._galaris_authorization_loops[run_id] = loop
        result, usage = await loop.run_in_executor(None, tracked_execution)"""
early_stop = """        if run_id in self._stopping_run_ids:
            _finish("cancelled")
            return"""
assert source.count(original) == 1, "Hermes worker boundary changed; review the stop evidence adapter."
assert source.count(early_stop) == 1, "Hermes early stop changed; review the stop evidence adapter."
terminal_extra = "        extra = extra or {}"
assert source.count(terminal_extra) == 1, "Hermes terminal envelope changed; review the stop evidence adapter."
source = source.replace(original, replacement).replace(
    early_stop, early_stop.replace('_finish("cancelled")', '_finish("cancelled", {"execution_stopped": True})')
).replace(
    terminal_extra,
    '        extra = {"execution_stopped": self._run_statuses.get(run_id, {}).get("execution_stopped") is True, **(extra or {})}',
)
approval_boundary = '        event = dict(approval_data or {})'
assert source.count(approval_boundary) == 1, "Review the exact Hermes approval payload boundary."
source = source.replace(approval_boundary, '''        event = dict(approval_data or {})
        import hashlib
        import json
        event["action_digest"] = hashlib.sha256(json.dumps(
            approval_data, sort_keys=True, separators=(",", ":"), default=str,
        ).encode()).hexdigest()
        event["allow_session"] = False
        event["allow_permanent"] = False''')
# Runtime APIs cannot grant a session, even if an old interaction sends that choice.
allowed_choices = '    allowed = {"once", "deny"} if room_scoped else {"once", "session", "always", "deny"}'
assert source.count(allowed_choices) == 1, "Review Hermes approval scope validation."
source = source.replace(allowed_choices, '    allowed = {"once", "deny"}')
source = source.replace('(room_scoped and resolve_all,', '(resolve_all,')
source = source.replace('(room_scoped and not request_id,', '(not request_id,')
managed_boundaries = {
    'def _http_routes(self) -> list[tuple[str, str, Any]]:\n    return [':
        'def _http_routes(self) -> list[tuple[str, str, Any]]:\n'
        '    async def managed_capabilities(request):\n'
        '        error = self._check_auth(request)\n'
        '        if error is not None:\n'
        '            return error\n'
        '        return web.json_response({"authorization_protocol": "galaris.runtime-authorization/v1", "resumable_runs": True})\n'
        '    return [("GET", "/v1/galaris/capabilities", managed_capabilities),',
    '    self._run_idempotency_ids: set[str] = set()':
        '    self._galaris_authorization_loops = {}\n    self._run_idempotency_ids: set[str] = set()',
    '    turn_author: Optional[Dict[str, Any]] = None':
        '    turn_author: Optional[Dict[str, Any]] = None\n    galaris_run_context: str = ""',
    '        turn_author=turn_author)':
        '        turn_author=turn_author, galaris_run_context=str(body.get("galaris_run_context") or ""))',
    '            register_gateway_notify(run.approval_session_key, approval_notify)':
        '            from galaris_authorization import bind_run, reset_run\n'
        '            resets.append((bind_run(self, run), reset_run))\n'
        '            register_gateway_notify(run.approval_session_key, approval_notify)',
}
for boundary, replacement in managed_boundaries.items():
    assert source.count(boundary) == 1, f"Review the Hermes managed context boundary: {boundary}"
    source = source.replace(boundary, replacement)
compile(source, str(path), "exec")
path.write_text(source)
